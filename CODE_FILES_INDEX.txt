================================================================================
USAGE EXAMPLES
================================================================================

Basic Usage:

  # Load and process single sample
  source("R/01_data_ingestion.R")
  result <- run_data_ingestion(
    vcf_s3_path = "s3://bucket/sample.vcf.gz",
    sample_id = "SAMPLE_001",
    ethnic_group = "EUR"
  )

  # Run quality control
  source("R/02_quality_control.R")
  qc_report <- run_quality_control(
    variants_df = result$variants,
    sample_id = "SAMPLE_001"
  )

  # Annotate variants
  source("R/03_variant_annotation.R")
  annotated <- annotate_variants(
    variants_df = result$variants,
    sample_id = "SAMPLE_001"
  )

Full Pipeline:

  # Execute complete pipeline
  Rscript scripts/run_pipeline.R SAMPLE_001 s3://bucket/sample.vcf.gz EUR

Docker Usage:

  # Build container
  docker build -t genomic-pipeline:latest .

  # Run with compose
  docker-compose up -d

  # Execute analysis
  docker-compose exec genomic-pipeline \
    Rscript scripts/run_pipeline.R SAMPLE_001 s3://bucket/file.vcf.gz EUR

QUICK START WITH CODE
================================================================================

Step 1: Load Libraries
  library(VariantAnnotation)
  library(tidyverse)
  library(paws)
  library(logger)

Step 2: Load Modules
  source("R/01_data_ingestion.R")
  source("R/02_quality_control.R")
  source("R/03_variant_annotation.R")

Step 3: Configure
  config <- yaml::read_yaml("config/pipeline_params.yml")

Step 4: Run Analysis
  # Data ingestion
  data <- run_data_ingestion(
    vcf_s3_path = "s3://bucket/sample.vcf.gz",
    sample_id = "SAMPLE_001",
    ethnic_group = "EUR"
  )
  
  # Quality control
  qc <- run_quality_control(
    variants_df = data$variants,
    sample_id = "SAMPLE_001"
  )
  
  # Annotation
  annotated <- annotate_variants(
    variants_df = data$variants,
    sample_id = "SAMPLE_001"
  )

Step 5: Export Results
  write.csv(annotated$variants, "results/annotated.csv")
